import pandas as pd
import json
from google.colab import files

# 1. Ανέβασμα αρχείου
print("Ανέβασε το αρχείο Excel:")
uploaded = files.upload()
file_name = list(uploaded.keys())[0]

# 2. Διάβασμα του Excel
df = pd.read_excel(file_name, header=1, dtype=object)

print(f"✅ Στήλες που εντοπίστηκαν: {list(df.columns)}")

json_list = []

for index, row in df.iterrows():
    # Παίρνουμε το Barcode
    barcode_raw = row.get('BARCODE')
    barcode_str = str(barcode_raw).strip() if pd.notna(barcode_raw) else ""

    # Έλεγχος αν περιέχει ψηφία (εγκυρότητα γραμμής)
    if any(char.isdigit() for char in barcode_str):

        # --- ΧΕΙΡΙΣΜΟΣ ΠΕΡΙΓΡΑΦΗΣ ---
        desc_raw = row.get('ΠΕΡΙΓΡΑΦΗ')
        if pd.isna(desc_raw):
            description_main = ""
        elif desc_raw == 1 or desc_raw == "1":
            description_main = "100%"
        else:
            description_main = str(desc_raw).strip()

        # --- ΧΕΙΡΙΣΜΟΣ ΒΑΡΟΥΣ ---
        weight_raw = row.get('ΒΑΡΟΣ (g)')
        weight_str = str(weight_raw).strip() if pd.notna(weight_raw) else ""

        # Ενοποίηση: Περιγραφή + Κενό + Βάρος
        # Αν το βάρος υπάρχει, το προσθέτουμε στο τέλος
        full_description = f"{description_main} {weight_str}ml".strip()

        # Καθαρισμός Τιμής
        try:
            price_val = row.get('ΤΙΜΗ', 0)
            price = float(str(price_val).replace(',', '.').replace('€', '').strip())
        except:
            price = 0.0

        item = {
            "Code": str(row.get('ΚΩΔΙΚΟΣ', '')).strip(),
            "Description": full_description, # Εδώ μπαίνει η ενοποιημένη περιγραφή
            "Brand": "ZUEGG",
            "Price": price,
            "ImageUrl": str(row.get('URL', '')).strip() if pd.notna(row.get('URL')) else ""
        }
        json_list.append(item)

# 3. Αποθήκευση
output_file = 'products_with_weight.json'
with open(output_file, 'w', encoding='utf-8') as f:
    json.dump(json_list, f, ensure_ascii=False, indent=2)

files.download(output_file)
print(f"--- ΟΛΟΚΛΗΡΩΘΗΚΕ ---")
print(f"Μετατράπηκαν {len(json_list)} προϊόντα με ενσωματωμένο βάρος στην περιγραφή.")
